vector<int> findSubstring(string s, vector<string>& words) {
    int w = words[0].length();
    int k = words.size();
    int total_len = k * w;
    vector<int> result;
    unordered_map<string, int> original_count;
    for (string word : words) {
        original_count[word]++;
    }
    for (int m = 0; m < w; ++m) {
        vector<string> words_list;
        for (int i = m; i + w <= s.length(); i += w) {
            words_list.push_back(s.substr(i, w));
        }
        if (words_list.size() < k) continue;
        unordered_map<string, int> current_count;
        for (int i = 0; i < k; ++i) {
            current_count[words_list[i]]++;
        }
        // check first window
        bool valid = true;
        for (auto& pair : original_count) {
            if (current_count.find(pair.first) == current_count.end() || current_count[pair.first] != pair.second) {
                valid = false;
                break;
            }
        }
        if (valid) {
            result.push_back(m);
        }
        for (int i = 1; i <= words_list.size() - k; ++i) {
            // remove the leftmost word
            string left = words_list[i - 1];
            current_count[left]--;
            if (current_count[left] == 0) {
                current_count.erase(left);
            }
            // add the new word
            string right = words_list[i + k - 1];
            current_count[right]++;
            // check validity
            valid = true;
            for (auto& pair : original_count) {
                if (current_count.find(pair.first) == current_count.end() || current_count[pair.first] != pair.second) {
                    valid = false;
                    break;
                }
            }
            if (valid) {
                result.push_back(m + i * w);
            }
        }
    }
    return result;
}
