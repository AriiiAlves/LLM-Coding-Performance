#include <vector>
using namespace std;

double findMedianSortedArrays(vector<int>& nums1, vector<int>& nums2) {
    int m = nums1.size();
    int n = nums2.size();
    
    if (m > n) {
        swap(nums1, nums2);
        swap(m, n);
    }
    
    int low = 0, high = m;
    int partition1, partition2;
    int max1 = INT_MIN, min2 = INT_MAX;
    
    while (low <= high) {
        partition1 = (low + high) / 2;
        partition2 = (m + n + 1) / 2 - partition1;
        
        max1 = (partition1 > 0 ? nums1[partition1 - 1] : INT_MIN);
        min2 = (partition2 < n ? nums2[partition2] : INT_MAX);
        
        if (max1 <= min2) {
            if (partition1 == 0 || partition2 == n || nums2[partition2 - 1] <= nums1[partition1 - 1]) {
                if ((m + n) % 2 == 0) {
                    return (max(max1, nums2[partition2 - 1]) + min(min2, nums1[partition1])) / 2.0;
                } else {
                    return max(max1, nums2[partition2 - 1]);
                }
            }
            low = partition1 + 1;
        } else {
            high = partition1 - 1;
        }
    }
    
    return 0.0;
}
